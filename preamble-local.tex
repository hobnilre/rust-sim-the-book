\newcommand{\bookpublicationdetails}{}

\renewcommand{\maketitle}{%
  \begin{titlepage}
    \centering
    \vspace*{0.7in}
    {\Huge\sffamily\bfseries\@title\par}
    \vspace{0.6in}
    {\large\@author\par}
    \vspace{0.3in}
    {\normalsize\@date\par}
    \vfill
    {\raggedright\scriptsize\bookpublicationdetails\par}
  \end{titlepage}%
}
